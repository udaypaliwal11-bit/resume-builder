        function adjustZoom(delta) {

        function autoFitPreviewZoom() {

            const viewport = document.getElementById('preview-viewport');

            if (!viewport) return;

            const availableWidth = viewport.clientWidth - 48;

            const a4PixelWidth = 794; // 210mm at 96dpi

            if (availableWidth > 300 && availableWidth < a4PixelWidth) {

                currentZoom = Math.max(0.42, +(availableWidth / a4PixelWidth).toFixed(2));

            } else if (availableWidth >= a4PixelWidth) {

                currentZoom = Math.min(0.92, +(availableWidth / (a4PixelWidth + 80)).toFixed(2));

            }

            applyZoom();

        }


        function applyZoom() {

            const sheet = document.getElementById('resume-sheet');

            const label = document.getElementById('zoom-label');

            if (sheet) {

                sheet.style.transform = `scale(${currentZoom})`;

                sheet.style.marginBottom = `${(currentZoom - 1) * 1122}px`;

            }

            if (label) {

                label.textContent = `${Math.round(currentZoom * 100)}%`;

            }

        }

                checks.push({ pass: true, title: "Complete Contact & Developer Links (+15)", desc: "Email, Phone, LinkedIn, and GitHub links detected for recruiter verification." });

            } else {

                totalScore += hasCoreContact ? 8 : 3;

                checks.push({ pass: false, title: "Missing Tech Profile Links (+7 potential)", desc: "Ensure both LinkedIn and GitHub URLs are filled in Personal Info." });

            }


            // 2. Categorized Technical Skills (20 pts)

            const s = resumeState.skills;

            const skillCategoriesCount = [s.languages, s.frameworks, s.cloudDevops, s.coreSystems].filter(v => v && v.trim().length > 3).length;

            if (skillCategoriesCount >= 3) {

                totalScore += 20;

                checks.push({ pass: true, title: "Structured ATS Skill Taxonomy (+20)", desc: `${skillCategoriesCount} distinct technical skill categories populated.` });

            } else {

                totalScore += skillCategoriesCount * 5;

                checks.push({ pass: false, title: "Expand Technical Skill Categories", desc: "Populate Languages, Frameworks, Cloud/DevOps, and Core CS for keyword matching." });

            }


            // 3. Quantified Impact Metrics (%, latency, scale numbers) (25 pts)

            const allBullets = [

                ...(resumeState.experience || []).flatMap(e => e.bullets || []),

                ...(resumeState.projects || []).flatMap(pr => pr.bullets || []),

                ...(resumeState.achievements || [])

            ];

            const metricRegex = /(\d+(?:\.\d+)?%|\d+(?:,\d+)+\+?|\$\d+|\d+x\b|\d+\s*(?:ms|sec|TPS|RPS|users|nodes|stars))/i;

            const quantifiedBullets = allBullets.filter(b => metricRegex.test(b));


            if (quantifiedBullets.length >= 4) {

                totalScore += 25;

                checks.push({ pass: true, title: `Strong Quantified Impact (${quantifiedBullets.length} metrics found) (+25)`, desc: "Bullets effectively use %, latency reductions, scale, or rankings." });

            } else {

                const pts = quantifiedBullets.length * 6;

                totalScore += pts;

                checks.push({ pass: false, title: `Add More Quantified Metrics (${quantifiedBullets.length}/4 minimum)`, desc: "Include concrete numbers (e.g., 'cut latency by 40%', '15,000 req/s', 'CGPA 9.1')." });

            }


            // 4. Strong Action Verbs (20 pts)

            let actionVerbCount = 0;

            allBullets.forEach(b => {

                const firstWord = b.trim().split(/\s+/)[0]?.toLowerCase().replace(/[^a-z]/g, '');

                if (STRONG_ACTION_VERBS.includes(firstWord)) actionVerbCount++;

            });


            if (actionVerbCount >= 4) {

                totalScore += 20;

                checks.push({ pass: true, title: `High-Impact Action Verbs (${actionVerbCount} bullets) (+20)`, desc: "Bullets lead with engineering verbs like Engineered, Architected, Optimized." });

            } else {

                totalScore += Math.min(15, actionVerbCount * 4);

                checks.push({ pass: false, title: "Start Bullets with Strong Action Verbs", desc: "Begin points with words like Engineered, Architected, Spearheaded, or Reduced." });

            }


            // 5. Mode-Specific Completeness Check (20 pts)

            if (resumeState.mode === 'student') {

                const hasEdu = resumeState.education && resumeState.education.length > 0;

                const hasProj = resumeState.projects && resumeState.projects.length >= 2;

                if (hasEdu && hasProj) {

                    totalScore += 20;

                    checks.push({ pass: true, title: "Student Mode Readiness (+20)", desc: "Education with CGPA/Score and 2+ technical projects with tech stacks verified." });

                } else {

                    totalScore += 10;

                    checks.push({ pass: false, title: "Student Mode Completeness", desc: "Add at least 2 technical projects with GitHub links and complete Education details." });

                }

            } else {

                const hasSummary = p.summary && p.summary.trim().length > 40;

                const hasExp = resumeState.experience && resumeState.experience.length >= 2;

                if (hasSummary && hasExp) {

                    totalScore += 20;

                    checks.push({ pass: true, title: "Tech Official Executive Readiness (+20)", desc: "Executive summary and multi-role leadership/architecture experience verified." });

                } else {

                    totalScore += 10;

                    checks.push({ pass: false, title: "Tech Official Completeness", desc: "Include an Executive Summary and at least 2 detailed work experience entries." });

                }

            }


            totalScore = Math.min(100, Math.max(10, totalScore));


            // Update UI Indicator

            const scoreVal = document.getElementById('ats-score-val');

            const badge = document.getElementById('ats-badge');

            const ring = document.getElementById('ats-score-ring');

            const feedback = document.getElementById('ats-feedback-text');


            if (scoreVal) scoreVal.textContent = `${totalScore}%`;


            if (totalScore >= 85) {

                ring.className = "relative w-11 h-11 rounded-full bg-slate-900 border-2 border-emerald-500 flex items-center justify-center shrink-0";

                scoreVal.className = "text-xs font-bold text-emerald-400";

                badge.className = "text-[10px] px-1.5 py-0.5 rounded bg-emerald-500/15 text-emerald-400 font-medium";

                badge.textContent = "ATS Optimized";

            } else if (totalScore >= 65) {

                ring.className = "relative w-11 h-11 rounded-full bg-slate-900 border-2 border-amber-500 flex items-center justify-center shrink-0";

                scoreVal.className = "text-xs font-bold text-amber-400";

                badge.className = "text-[10px] px-1.5 py-0.5 rounded bg-amber-500/15 text-amber-400 font-medium";

                badge.textContent = "Good — Can Improve";

            } else {

                ring.className = "relative w-11 h-11 rounded-full bg-slate-900 border-2 border-rose-500 flex items-center justify-center shrink-0";

                scoreVal.className = "text-xs font-bold text-rose-400";

                badge.className = "text-[10px] px-1.5 py-0.5 rounded bg-rose-500/15 text-rose-400 font-medium";

                badge.textContent = "Needs Attention";

            }


            const failedCheck = checks.find(c => !c.pass);

            if (feedback) {

                feedback.textContent = failedCheck

                    ? `Tip: ${failedCheck.desc}`

                    : `All ${checks.length} ATS parser & quantified impact checks passed!`;

            }


            // Populate Audit Modal

            const modalBody = document.getElementById('ats-modal-body');

            if (modalBody) {

                modalBody.innerHTML = checks.map(c => `

                    <div class="p-3 rounded-xl border ${c.pass ? 'bg-emerald-950/20 border-emerald-800/50' : 'bg-amber-950/20 border-amber-800/50'} flex items-start gap-3">

                        <i class="fa-solid ${c.pass ? 'fa-circle-check text-emerald-400' : 'fa-triangle-exclamation text-amber-400'} mt-0.5"></i>

                        <div>

                            <div class="font-semibold text-white">${c.title}</div>

                            <div class="text-slate-400 mt-0.5">${c.desc}</div>

                        </div>

                    </div>

                `).join('');

            }

        }

        }


        function importResumeJSON(event) {

            const file = event.target.files[0];

            if (!file) return;

            const reader = new FileReader();

            reader.onload = function(e) {

                try {

                    const parsed = JSON.parse(e.target.result);

                    if (parsed && parsed.personal) {

                        resumeState = parsed;

                        syncUIWithState();

                        renderEditorForm();

                        renderResumePreview();

                        calculateATSScore();

                        saveState();

                        showToast("Imported Resume JSON successfully!");

                    } else {

                        showToast("Invalid Resume JSON structure", "fa-circle-exclamation text-rose-400");

                    }

                } catch (err) {

                    showToast("Error parsing JSON file", "fa-circle-exclamation text-rose-400");

                }

            };

            reader.readAsText(file);

            event.target.value = '';

        }

    </script>
